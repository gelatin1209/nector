function(nector_add_third_party name)
    add_library(${name} OBJECT ${ARGN})
    target_include_directories(${name} PUBLIC ${nector_third_party_includes})
    target_compile_options(${name} PRIVATE -w)
endfunction()
nector_add_third_party(nector_zlib ${zlib_sources})
nector_add_third_party(nector_brotli ${brotli_sources})
nector_add_third_party(nector_png ${png_sources})
nector_add_third_party(nector_jpeg ${jpeg_sources})
nector_add_third_party(nector_miniaudio src/audio/miniaudio_backend.cpp)

set(nector_public_includes "${CMAKE_CURRENT_SOURCE_DIR}/include" ${nector_third_party_includes})
set(nector_warning_flags -Wall -Wextra -Wpedantic -Werror)
set(nector_modules core rasterization font image nimation input scene audio platform interface)
foreach(module IN LISTS nector_modules)
    file(GLOB module_sources CONFIGURE_DEPENDS "src/${module}/*.cpp")
    if(module STREQUAL "audio")
        list(FILTER module_sources EXCLUDE REGEX "/miniaudio_backend\\.cpp$")
    endif()
    if(APPLE AND module STREQUAL "platform")
        file(GLOB cocoa_sources CONFIGURE_DEPENDS src/platform/*.mm)
        list(APPEND module_sources ${cocoa_sources})
    endif()
    add_library(nector_${module} SHARED ${module_sources})
    target_include_directories(nector_${module} PUBLIC ${nector_public_includes})
    set(export_name "${module}")
    if(module STREQUAL "nimation")
        set(export_name "anim")
    endif()
    target_compile_definitions(nector_${module} PRIVATE nector_${export_name}_export)
    target_compile_options(nector_${module} PRIVATE ${nector_warning_flags})
    string(SUBSTRING "${module}" 0 1 module_first)
    string(TOUPPER "${module_first}" module_first)
    string(SUBSTRING "${module}" 1 -1 module_rest)
    set(module_output "Nector.${module_first}${module_rest}")
    if(module STREQUAL "nimation")
        set(module_output "NImation")
    endif()
    set_target_properties(nector_${module} PROPERTIES OUTPUT_NAME "${module_output}")
    if(WIN32)
        set_target_properties(nector_${module} PROPERTIES PREFIX "" IMPORT_PREFIX "")
    endif()
endforeach()

set(nector_all_sources)

    file(GLOB module_sources CONFIGURE_DEPENDS "src/${module}/*.cpp")
    if(module STREQUAL "audio")
        list(FILTER module_sources EXCLUDE REGEX "/miniaudio_backend\\.cpp$")
    endif()
    if(APPLE AND module STREQUAL "platform")
        file(GLOB cocoa_sources CONFIGURE_DEPENDS src/platform/*.mm)
